10 REM UART1600 TEST V1.0: TX AND RX TEST OF UART1600 (19200 BPS 8N1)

40 A=0:INPUT "ML ADDRESS (ENTER=49349) ";A

100 REM --- MENU ---

120 IF Y$="1" THEN GOTO 200

320 PRINT "255 X 55H, STOP: BREAK"

400 REM --- RX: PACKETS ---

420 PRINT "WAITING FOR DATA..."

450 BEEP 1:PRINT "RX ";D;" BYTES:"

480 M=D:CALL S,M
500 Z$=""
510 FOR J=0 TO D-1:C=PEEK(R+J)

550 IF LEN(Z$)>=26 THEN PRINT Z$:Z$=""
560 NEXT J

580 GOTO 420
